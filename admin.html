🔧 Código Apps Script (envío automático)

Agrega esta función a tu Apps Script para habilitar el envío automático:

function enviarCorreoCampana(datos) {
  var email   = datos.email   || '';
  var nombre  = datos.nombre  || 'Cliente';
  var asunto  = datos.asunto  || '¡Tu multa puede eliminarse!';
  var cuenta  = datos.cuenta  || Session.getActiveUser().getEmail();

  if(!email || !email.includes('@')) return {ok:false,msg:'Email inválido'};

  var html = HtmlService.createHtmlOutput(datos.htmlBody || generarHtmlEmail(nombre)).getContent();

  try {
    GmailApp.sendEmail(email, asunto, '', {
      htmlBody: datos.htmlBody || generarHtmlEmail(nombre),
      name: datos.nombreRemitente || 'Jurídicos Web',
      from: cuenta
    });
    return {ok:true};
  } catch(e) {
    return {ok:false, msg: e.message};
  }
}
