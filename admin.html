Cuentas Gmail adicionales

Cada cuenta debe tener su propio Apps Script desplegado. El sistema rota cuando Brevo se agota.

📋 ¿Cómo obtener la URL del script de cada cuenta?
Para CADA cuenta Gmail adicional:
1. Inicia sesión en Gmail con esa cuenta
2. Ve a script.google.com → Nuevo proyecto
3. Borra el código de ejemplo y pega el Script Secundario de abajo
4. Implementar → Nueva implementación → Tipo: Aplicación web
  · Ejecutar como: Yo (esa cuenta)
  · Acceso: Cualquier usuario
5. Copia la URL y pégala arriba en "URL del script"

Script secundario (copiar en cada cuenta Gmail)

/* ── SCRIPT SECUNDARIO JurídicosWeb ──────────────────────────
   Despliega ESTE script en cada cuenta Gmail adicional.
   Una vez desplegado como "Aplicación web → Cualquier usuario",
   copia la URL y pégala en admin.html → Cuentas → URL del script.
   ─────────────────────────────────────────────────────────── */

var EMAIL_DESPACHO = 'juridicoswebcasos@gmail.com';

function doPost(e) {
  try {
    var datos = JSON.parse(e.postData.contents);
    var accion = datos._accion || datos.accion || '';

    if (accion === 'test_campana') {
      return json({ ok: true, gasOk: true, version: '2026-10',
                    cuenta: Session.getEffectiveUser().getEmail() });
    }

    if (accion !== 'enviar_correo') {
      return json({ ok: false, msg: 'Accion no soportada: ' + accion });
    }

    var email  = datos.email  || '';
    var asunto = datos.asunto || '';
    var html   = datos.html   || '';
    var nombre = datos.nombreRemitente || 'JurídicosWeb';

    if (!email || email.indexOf('@') < 0) {
      return json({ ok: false, msg: 'Email inválido' });
    }

    GmailApp.sendEmail(email, asunto,
      'Por favor activa HTML en tu cliente de correo para ver este mensaje.',
      {
        htmlBody: html,
        name: nombre,
        replyTo: EMAIL_DESPACHO
      }
    );
    return json({ ok: true, via: 'gmail',
                  desde: Session.getEffectiveUser().getEmail() });

  } catch (ex) {
    var msg = ex.message || String(ex);
    var quota = /demasiadas|too many|quota|daily|limit/i.test(msg);
    return json({ ok: false, msg: msg, quotaExcedida: quota });
  }
}

function doGet(e) {
  return json({ ok: true, gasOk: true, version: '2026-10',
                cuenta: Session.getEffectiveUser().getEmail() });
}

function json(obj) {
  return ContentService
    .createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

function enviarCorreoCampana(datos) {
  var email  = datos.email  || '';
  var asunto = datos.asunto || 'Tu multa de transito tiene solucion legal';
  if (!email || !email.includes('@'))
    return respuestaJson({ ok: false, msg: 'Email invalido' });
  // Sin "from:" — siempre funciona y siempre queda en Enviados
  try {
    GmailApp.sendEmail(email, asunto,
      'Para ver este correo activa HTML en tu cliente de correo.',
      {
        htmlBody: datos.html || '',
        name: datos.nombreRemitente || 'JuridicosWeb',
        replyTo: 'juridicoswebcasos@gmail.com'
      }
    );
    return respuestaJson({ ok: true });
  } catch (e) {
    try {
      MailApp.sendEmail({
        to: email, subject: asunto,
        htmlBody: datos.html || '',
        name: datos.nombreRemitente || 'JuridicosWeb',
        replyTo: 'juridicoswebcasos@gmail.com'
      });
      return respuestaJson({ ok: true, via: 'mailapp' });
    } catch (e2) {
      return respuestaJson({ ok: false, msg: e2.message });
    }
  }
}

var SYNC_SHEET_NAME = 'CampanSync';

function syncCampanaSave(datos) {
  var ss = SpreadsheetApp.openById(SHEET_ID);
  var sh = ss.getSheetByName(SYNC_SHEET_NAME) || ss.insertSheet(SYNC_SHEET_NAME);
  sh.clearContents();
  sh.getRange(1, 1).setValue(datos.data || '{}');
  sh.getRange(1, 2).setValue(new Date().toISOString());
  return respuestaJson({ ok: true });
}

function syncCampanaLoad() {
  var ss = SpreadsheetApp.openById(SHEET_ID);
  var sh = ss.getSheetByName(SYNC_SHEET_NAME);
  if (!sh || sh.getLastRow() < 1) return respuestaJson({ ok: true, data: null });
  var raw = sh.getRange(1, 1).getValue();
  return respuestaJson({ ok: true, data: raw || null });
}
